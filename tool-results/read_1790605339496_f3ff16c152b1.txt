    33→// ─────────────────────────────────────────────────────────────────────────────
    34→// Premium shader-driven starfield.
    35→// Each star has its own brightness, twinkle phase/frequency, and stellar
    36→// classification color (blue / white / yellow-white / orange / red).
    37→// Vertex shader does perspective size attenuation; fragment shader renders
    38→// a soft circular point with smooth falloff + bright core.
    39→// ─────────────────────────────────────────────────────────────────────────────
    40→const STAR_VERTEX = /* glsl */ `
    41→  attribute float aBrightness;
    42→  attribute float aPhase;
    43→  attribute float aFreq;
    44→  attribute vec3 aColor;
    45→
    46→  uniform float uTime;
    47→  uniform float uPixelRatio;
    48→
    49→  varying float vBrightness;
    50→  varying vec3 vColor;
    51→
    52→  void main() {
    53→    vColor = aColor;
    54→    // Twinkle: base brightness modulated by a per-star sinusoid.
    55→    // Range stays in [0.4, 1.0] of base brightness — never fully extinguished.
    56→    float twinkle = 0.7 + 0.3 * sin(uTime * aFreq + aPhase);
    57→    vBrightness = aBrightness * twinkle;
    58→
    59→    vec4 mvPosition = modelViewMatrix * vec4(position, 1.0);
    60→    gl_Position = projectionMatrix * mvPosition;
    61→
    62→    // Size attenuation: closer stars render larger. Brighter stars render
    63→    // larger too — twinkling visibly grows the point as it peaks.
    64→    float baseSize = 1.6 + aBrightness * 4.2;
    65→    float dist = max(1.0, -mvPosition.z);
    66→    gl_PointSize = baseSize * uPixelRatio * (240.0 / dist);
    67→  }
    68→`;
    69→
    70→const STAR_FRAGMENT = /* glsl */ `
    71→  varying float vBrightness;
    72→  varying vec3 vColor;
    73→
    74→  void main() {
    75→    // gl_PointCoord is [0,1] across the point sprite. Center it.
    76→    vec2 uv = gl_PointCoord - 0.5;
    77→    float dist = length(uv);
    78→    if (dist > 0.5) discard;
    79→
    80→    // Soft circular falloff + a brighter tight core for stellar sparkle.
    81→    float falloff = smoothstep(0.5, 0.0, dist);
    82→    float core = smoothstep(0.28, 0.0, dist);
    83→    vec3 col = vColor * vBrightness * (0.55 + 0.6 * core);
    84→    float alpha = falloff * clamp(vBrightness, 0.0, 1.0);
    85→
    86→    gl_FragColor = vec4(col, alpha);
    87→  }
    88→`;
    89→
    90→// ─────────────────────────────────────────────────────────────────────────────
    91→// Chromatic aberration + vignette (single combined pass — cheap).
    92→// CA: radial RGB channel offset, zero at center, max ~1.5px at corners.
    93→// Vignette: smooth radial darkening, center = full, corners = ~0.75.
    94→// ─────────────────────────────────────────────────────────────────────────────
    95→const CA_VIGNETTE_SHADER = {
    96→  uniforms: {
    97→    tDiffuse: { value: null as THREE.Texture | null },
    98→    uIntensity: { value: 0.4 },
    99→    uVignette: { value: 0.25 },
   100→  },
   101→  vertexShader: /* glsl */ `
   102→    varying vec2 vUv;
   103→    void main() {
   104→      vUv = uv;
   105→      gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);
   106→    }
   107→  `,
   108→  fragmentShader: /* glsl */ `
   109→    uniform sampler2D tDiffuse;
   110→    uniform float uIntensity;
   111→    uniform float uVignette;
   112→    varying vec2 vUv;
   113→
   114→    void main() {
   115→      vec2 uv = vUv;
   116→      vec2 offset = uv - vec2(0.5);
   117→      float dist = length(offset);
   118→
   119→      // Radial falloff: dist^2 so center is clean, edges ramp up.
   120→      float caAmt = uIntensity * 0.012 * dist * dist;
   121→      vec3 color;
   122→      if (caAmt > 1e-6) {
   123→        vec2 dir = offset / max(dist, 1e-5);
   124→        // R shifted outward toward corner; B shifted inward toward center.
   125→        color.r = texture2D(tDiffuse, uv - dir * caAmt).r;
   126→        color.g = texture2D(tDiffuse, uv).g;
   127→        color.b = texture2D(tDiffuse, uv + dir * caAmt).b;
   128→      } else {
   129→        color = texture2D(tDiffuse, uv).rgb;
   130→      }
   131→
   132→      // Soft cinematic vignette. Center stays at 1.0, corners ~0.75.
   133→      float vignette = 1.0 - uVignette * dist * dist;
   134→      color *= vignette;
   135→
   136→      gl_FragColor = vec4(color, 1.0);
   137→    }
   138→  `,
   139→};
   140→
   141→// ─────────────────────────────────────────────────────────────────────────────
   142→// Film grain: animated hash-based noise overlay. VERY subtle — adds organic
   143→// texture like Apple's HDR pipeline. Operates per-pixel; high/ultra only.
   144→// ─────────────────────────────────────────────────────────────────────────────
   145→const FILM_GRAIN_SHADER = {
   146→  uniforms: {
   147→    tDiffuse: { value: null as THREE.Texture | null },
   148→    uTime: { value: 0 },
   149→    uResolution: { value: new THREE.Vector2(1, 1) },
   150→    uAmount: { value: 0.04 },
   151→  },
   152→  vertexShader: /* glsl */ `
   153→    varying vec2 vUv;
   154→    void main() {
   155→      vUv = uv;
   156→      gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);
   157→    }
   158→  `,
   159→  fragmentShader: /* glsl */ `
   160→    uniform sampler2D tDiffuse;
   161→    uniform float uTime;
   162→    uniform vec2 uResolution;
   163→    uniform float uAmount;
   164→    varying vec2 vUv;
   165→
   166→    // Cheap hash-based noise: deterministic per cell, fully GPU-side.
   167→    float hash(vec2 p) {
   168→      p = fract(p * vec2(123.34, 456.21));
   169→      p += dot(p, p + 45.32);
   170→      return fract(p.x * p.y);
   171→    }
   172→
   173→    void main() {
   174→      vec3 color = texture2D(tDiffuse, vUv).rgb;
   175→      // Animated grain — changes every frame, varies spatially per pixel.
   176→      float noise = hash(vUv * uResolution + uTime * 53.0);
   177→      // Subtle modulation (±0.02 in linear). Apple-HDR-style texture.
   178→      color += (noise - 0.5) * uAmount;
   179→      gl_FragColor = vec4(color, 1.0);
   180→    }
   181→  `,
   182→};

   197→/** Orbit/pan/zoom camera rig (mouse, touch-drag, wheel, arrow keys). */
   198→export class CameraRig {
   199→  readonly target = new THREE.Vector3();
   200→  yaw = 0;
   201→  pitch = 0.4;
   202→  distance = 13;
   203→  autoRotate = false;
   204→  private readonly home = { yaw: 0, pitch: 0.4, distance: 13, target: new THREE.Vector3() };
   205→  private readonly tmpOffset = new THREE.Vector3();
   206→
   207→  setHome(view: WorldView): void {
   208→    this.home.yaw = view.yaw;
   209→    this.home.pitch = view.pitch;
   210→    this.home.distance = view.distance;
   211→    this.home.target.set(0, 0, 0);
   212→    this.resetToHome();
   213→  }
   214→
   215→  resetToHome(): void {
   216→    this.yaw = this.home.yaw;
   217→    this.pitch = this.home.pitch;
   218→    this.distance = this.home.distance;
   219→    this.target.copy(this.home.target);
   220→  }
   221→
   222→  orbitBy(dxPixels: number, dyPixels: number): void {
   223→    const s = PrismConfig.cameraRig.orbitSpeed;
   224→    this.yaw -= dxPixels * s;
   225→    this.pitch = THREE.MathUtils.clamp(this.pitch + dyPixels * s, 0.03, 1.52);
   226→  }
   227→
   228→  dolly(factor: number): void {
   229→    const c = PrismConfig.cameraRig;
   230→    this.distance = THREE.MathUtils.clamp(this.distance * factor, c.minDistance, c.maxDistance);
   231→  }
   232→
   233→  panBy(dxPixels: number, dyPixels: number, camera: THREE.Camera): void {
   234→    const scale = (this.distance * PrismConfig.cameraRig.panSpeed) / 10;
   235→    const right = this.tmpOffset.setFromMatrixColumn(camera.matrix, 0);
   236→    const up = new THREE.Vector3().setFromMatrixColumn(camera.matrix, 1);
   237→    this.target.addScaledVector(right, -dxPixels * scale).addScaledVector(up, dyPixels * scale);
   238→    this.target.y = THREE.MathUtils.clamp(this.target.y, -6, 6);
   239→  }
   240→
   241→  setView(name: 'overview' | 'top' | 'edge'): void {
   242→    if (name === 'top') {
   243→      this.pitch = 1.5;
   244→      this.yaw = 0;
   245→    } else if (name === 'edge') {
   246→      this.pitch = 0.06;
   247→    } else {
   248→      this.resetToHome();
   249→    }
   250→  }
   251→
   252→  /** Impact shake 0..1 (decays exponentially, squared falloff settles soft). */
   253→  private trauma = 0;
   254→  /** Accessibility gate: reduced-motion users get a whisper of shake. */
   255→  shakeScale = 1;
   256→
   257→  addShake(amount: number): void {
   258→    this.trauma = Math.min(1, this.trauma + Math.max(0, amount));
   259→  }
   260→
   261→  get traumaLevel(): number {
   262→    return this.trauma;
   263→  }
   264→
   265→  update(dt: number, camera: THREE.PerspectiveCamera): void {
   266→    if (this.autoRotate) this.yaw += dt * PrismConfig.cameraRig.autoRotateSpeed;
   267→    // NATURAL STEADICAM: rate 10 = ~15% per frame at 60fps. Responsive enough
   268→    // to feel direct, with just enough inertia for smooth motion. Was 8
   269→    // (slightly laggy on orbit stop).
   270→    const cp = Math.cos(this.pitch);
   271→    const targetX = this.target.x + this.distance * cp * Math.sin(this.yaw);
   272→    const targetY = this.target.y + this.distance * Math.sin(this.pitch);
   273→    const targetZ = this.target.z + this.distance * cp * Math.cos(this.yaw);
   274→    const damp = 1 - Math.exp(-dt * 10);
   275→    camera.position.x += (targetX - camera.position.x) * damp;
   276→    camera.position.y += (targetY - camera.position.y) * damp;
   277→    camera.position.z += (targetZ - camera.position.z) * damp;
   278→    if (this.trauma > 0) {
   279→      const s = this.trauma * this.trauma * 0.4 * this.shakeScale;
   280→      camera.position.x += (Math.random() - 0.5) * 2 * s;
   281→      camera.position.y += (Math.random() - 0.5) * 2 * s;
   282→      camera.position.z += (Math.random() - 0.5) * 2 * s;
   283→      this.trauma *= Math.exp(-dt * 2.2);
   284→      if (this.trauma <= 0.003) this.trauma = 0;
   285→    }
   286→    camera.lookAt(this.target);
   287→  }
   288→}

   321→  constructor(container: HTMLElement, quality: QualityTier, preset: PresetId) {
   322→    this.quality = quality;
   323→    // Performance: request high-performance GPU adapter explicitly. This forces
   324→    // the browser to use the discrete GPU on dual-GPU machines (common on laptops).
   325→    // Use antialias only on high/ultra (MSAA is expensive; pixelRatio covers it on lower tiers).
   326→    const useAA = quality === 'high' || quality === 'ultra';
   327→    this.renderer = new THREE.WebGLRenderer({
   328→      antialias: useAA,
   329→      powerPreference: 'high-performance',
   330→      stencil: false,
   331→    });
   332→    this.renderer.setSize(container.clientWidth, container.clientHeight);
   333→    this.renderer.toneMapping = THREE.ACESFilmicToneMapping;
   334→    this.renderer.toneMappingExposure = 1.15;
   335→    // Performance: disable shadow maps (not used; saves allocation + render pass).
   336→    this.renderer.shadowMap.enabled = false;
   337→    this.applyPixelRatio();
   338→    container.appendChild(this.renderer.domElement);
   339→    // Screen readers get a labeled image role instead of a silent canvas.
   340→    this.renderer.domElement.setAttribute('role', 'img');
   341→    this.renderer.domElement.setAttribute(
   342→      'aria-label',
   343→      'Interactive 3D scene. Move the mouse to point, hold to grab, or enable the camera for hand control. Press H for all controls.',
   344→    );
   345→
   346→    this.scene.background = new THREE.Color(0x04060d);
   347→    this.scene.fog = new THREE.Fog(0x04060d, 20, 70);
   348→
   349→    this.camera = new THREE.PerspectiveCamera(
   350→      55,
   351→      container.clientWidth / Math.max(1, container.clientHeight),
   352→      0.1,
   353→      300,
   354→    );
   355→    this.camera.position.set(0, 4.6, 12);
   356→
   357→    // Multi-light setup (inspired by sanderblue/solar-system-threejs, Apache 2.0):
   358→    // OPTIMIZED: reduced from 4 directional lights to 2 (key + fill) — each
   359→    // light adds a full render pass for every MeshStandardMaterial. 2 lights
   360→    // + ambient covers all angles with half the cost.
   361→    this.scene.add(new THREE.AmbientLight(0xbfd4ff, 0.5));
   362→    const key = new THREE.DirectionalLight(0xffffff, 0.8);
   363→    key.position.set(3, 5, 4);
   364→    this.scene.add(key);
   365→    const fill = new THREE.DirectionalLight(0x8899bb, 0.3);
   366→    fill.position.set(-3, 2, -4);
   367→    this.scene.add(fill);
   368→
   369→    // Shared resources (built once; worlds must never dispose these).
   370→    this.glowTex = buildGlowTexture();
   371→    this.nebulaTex = buildNebulaTexture();
   372→    this.planetTex = buildPlanetTextures();
   373→    this.buildStarfield();
   374→
   375→    this.nebula = new THREE.Mesh(
   376→      new THREE.SphereGeometry(120, 24, 16),
   377→      new THREE.MeshBasicMaterial({ map: this.nebulaTex, side: THREE.BackSide, depthWrite: false, fog: false }),
   378→    );
   379→    this.nebula.visible = false;
   380→    this.scene.add(this.nebula);
   381→
   382→    // Hand cursor + pointer ray.
   383→    this.cursorMat = new THREE.MeshBasicMaterial({ color: 0x9adcff, transparent: true, opacity: 0.95 });
   384→    this.cursor = new THREE.Mesh(new THREE.OctahedronGeometry(0.09), this.cursorMat);
   385→    this.cursor.visible = false;
   386→    this.scene.add(this.cursor);
   387→
   388→    // Pinch progress ring: wide/faint when open, cinching onto the cursor
   389→    // as the pinch closes. Billboards toward the camera every frame.
   390→    this.cursorRingMat = new THREE.MeshBasicMaterial({
   391→      color: 0x9adcff,
   392→      transparent: true,
   393→      opacity: 0.25,
   394→      side: THREE.DoubleSide,
   395→      depthTest: false,
   396→    });
   397→    this.cursorRing = new THREE.Mesh(new THREE.RingGeometry(0.13, 0.16, 24), this.cursorRingMat);
   398→    this.cursorRing.visible = false;
   399→    this.cursorRing.renderOrder = 5;
   400→    this.scene.add(this.cursorRing);
   401→
   402→    this.rayPositions = new Float32Array(6);
   403→    const rayGeo = new THREE.BufferGeometry();
   404→    rayGeo.setAttribute('position', new THREE.BufferAttribute(this.rayPositions, 3));
   405→    this.rayLine = new THREE.Line(
   406→      rayGeo,
   407→      new THREE.LineBasicMaterial({ color: 0x9adcff, transparent: true, opacity: 0.35 }),
   408→    );
   409→    this.rayLine.visible = false;
   410→    this.rayLine.frustumCulled = false;
   411→    this.scene.add(this.rayLine);
   412→
   413→    this.scene.add(this.world);
   414→    this.scene.add(this.labelLayer);
   415→
   416→    // Post-processing composer. Pass order is critical for the cinematic look:
   417→    //   RenderPass → UnrealBloomPass → ChromaticAberration+Vignette → FilmGrain → OutputPass
   418→    // Bloom lifts emissive bodies; CA + vignette add cinematic depth; grain
   419→    // adds organic texture; OutputPass applies tone mapping + color space.
   420→    // All passes are always constructed (cheap to instantiate); only
   421→    // composer.render() is gated by quality tier in render().
   422→    this.composer = new EffectComposer(this.renderer);
   423→    this.composer.addPass(new RenderPass(this.scene, this.camera));
   424→    const bloom = new UnrealBloomPass(
   425→      new THREE.Vector2(container.clientWidth, container.clientHeight),
   426→      PrismConfig.bloom.strength,
   427→      PrismConfig.bloom.radius,
   428→      PrismConfig.bloom.threshold,
   429→    );
   430→    this.composer.addPass(bloom);
   431→    this.caPass = new ShaderPass(CA_VIGNETTE_SHADER);
   432→    this.composer.addPass(this.caPass);
   433→    this.composer.addPass(new OutputPass());
   434→    // NOTE: the FILM_GRAIN_SHADER pass was removed — its per-pixel hash noise
   435→    // animated every frame and read as 'TV static covering the whole 3D scene'
   436→    // even at uAmount = 0.04. Not premium; just noisy. The vignette in CA_VIGNETTE
   437→    // is enough cinematic depth.
   438→
   439→    this.loadPreset(preset);
   440→  }

   471→  loadPreset(id: PresetId): void {
   472→    if (this.api) {
   473→      this.api.dispose();
   474→      this.world.clear();
   475→      this.labelLayer.clear();
   476→    }
   477→    this.presetId = id;
   478→    // Two-hand transform must not leak between worlds.
   479→    this.world.scale.setScalar(1);
   480→    this.world.rotation.set(0, 0, 0);
   481→    this.world.position.set(0, 0, 0);
   482→    const builder = BUILDERS[id];
   483→    this.api = builder({
   484→      world: this.world,
   485→      labelLayer: this.labelLayer,
   486→      glowTex: this.glowTex,
   487→      nebulaTex: this.nebulaTex,
   488→      planetTex: this.planetTex,
   489→      shakeCamera: (amount: number) => this.shakeCamera(amount),
   490→      quality: this.quality,
   491→    });
   492→    const bg = this.api.background;
   493→    if (bg === 'nebula') {
   494→      this.scene.background = null;
   495→      this.nebula.visible = true;
   496→    } else {
   497→      this.nebula.visible = false;
   498→      this.scene.background = new THREE.Color(bg);
   499→    }
   500→    this.stars.visible = this.api.stars ?? true;
   501→    this.rig.setHome(this.api.view);
   502→    this.setHover(null);
   503→    this.applyTextureMaps();
   504→  }

   522→    const ratio = PrismConfig.quality[this.quality]?.pixelRatio ?? 1.5;
   523→    const pr = Math.min(window.devicePixelRatio || 1, ratio);
   524→    this.renderer.setPixelRatio(pr);

   532→  private applyTextureMaps(): void {
   533→    const on = this.quality === 'high' || this.quality === 'ultra';
   534→    this.world.traverse((obj) => {

   537→      const texMap = mat?.userData.texMap as THREE.Texture | undefined;
   538→      if (mat && texMap) {
   539→        mat.map = on ? texMap : null;
   540→        mat.needsUpdate = true;
   541→      }
   542→    });
   543→  }

   560→  /**
   561→   * Premium shader-driven starfield. Each star carries per-vertex attributes
   562→   * (brightness, twinkle phase/freq, stellar-class color) feeding a custom
   563→   * ShaderMaterial. Count scales with quality tier. Always rendered (every
   564→   * tier) — the shader is essentially free; the cost is the point count.
   565→   */
   566→  private buildStarfield(): void {
   567→    const tier = this.quality;
   568→    const count = tier === 'ultra' ? 1500 : tier === 'high' ? 1000 : tier === 'medium' ? 600 : 400;
   569→    const positions = new Float32Array(count * 3);
   570→    const brightness = new Float32Array(count);
   571→    const phase = new Float32Array(count);
   572→    const freq = new Float32Array(count);
   573→    const color = new Float32Array(count * 3);
   574→
   575→    // Stellar classification distribution (approximate, tuned for visual
   576→    // variety rather than astrophysical accuracy):
   577→    //   ~15% blue (hot O/B), 20% white (A), 30% yellow-white (F/G),
   578→    //   25% orange (K), 10% red (M).
   579→    const classes: Array<{ weight: number; rgb: [number, number, number] }> = [
   580→      { weight: 0.15, rgb: [0.62, 0.72, 1.0] }, // blue
   581→      { weight: 0.2, rgb: [1.0, 1.0, 1.0] }, // white
   582→      { weight: 0.3, rgb: [1.0, 0.97, 0.85] }, // yellow-white
   583→      { weight: 0.25, rgb: [1.0, 0.78, 0.55] }, // orange
   584→      { weight: 0.1, rgb: [1.0, 0.55, 0.42] }, // red
   585→    ];
   586→
   587→    for (let i = 0; i < count; i++) {
   588→      // Spherical shell distribution: stars sit on a 60–120 unit sphere so
   589→      // they're always behind world geometry but inside the far plane (300).
   590→      const r = 60 + Math.random() * 60;
   591→      const theta = Math.random() * Math.PI * 2;
   592→      const phi = Math.acos(2 * Math.random() - 1);
   593→      positions[i * 3] = r * Math.sin(phi) * Math.cos(theta);
   594→      positions[i * 3 + 1] = r * Math.cos(phi);
   595→      positions[i * 3 + 2] = r * Math.sin(phi) * Math.sin(theta);
   596→
   597→      // Brightness in [0.3, 1.0] — biases toward dim stars (looks more
   598→      // realistic; few bright, many faint).
   599→      brightness[i] = 0.3 + Math.random() * 0.7;
   600→      phase[i] = Math.random() * Math.PI * 2;
   601→      // Twinkle frequency 0.5–3.0 Hz (slow drift to fast shimmer).
   602→      freq[i] = 0.5 + Math.random() * 2.5;
   603→
   604→      // Weighted random pick of stellar class.
   605→      const roll = Math.random();
   606→      let acc = 0;
   607→      let chosen = classes[2]; // default yellow-white
   608→      for (const c of classes) {
   609→        acc += c.weight;
   610→        if (roll <= acc) {
   611→          chosen = c;
   612→          break;
   613→        }
   614→      }
   615→      color[i * 3] = chosen.rgb[0];
   616→      color[i * 3 + 1] = chosen.rgb[1];
   617→      color[i * 3 + 2] = chosen.rgb[2];
   618→    }
   619→
   620→    const geo = new THREE.BufferGeometry();
   621→    geo.setAttribute('position', new THREE.BufferAttribute(positions, 3));
   622→    geo.setAttribute('aBrightness', new THREE.BufferAttribute(brightness, 1));
   623→    geo.setAttribute('aPhase', new THREE.BufferAttribute(phase, 1));
   624→    geo.setAttribute('aFreq', new THREE.BufferAttribute(freq, 1));
   625→    geo.setAttribute('aColor', new THREE.BufferAttribute(color, 3));
   626→
   627→    this.starMat = new THREE.ShaderMaterial({
   628→      uniforms: {
   629→        uTime: { value: 0 },
   630→        uPixelRatio: { value: this.renderer.getPixelRatio() },
   631→      },
   632→      vertexShader: STAR_VERTEX,
   633→      fragmentShader: STAR_FRAGMENT,
   634→      transparent: true,
   635→      depthWrite: false,
   636→      // Additive: stars never darken what's behind them, they only brighten —
   637→      // so overlapping stars sum into a believable bright cluster.
   638→      blending: THREE.AdditiveBlending,
   639→    });
   640→    this.stars = new THREE.Points(geo, this.starMat);
   641→    this.scene.add(this.stars);
   642→  }

   644→  /** Highlight the hovered body; previous hover is always restored. */
   645→  setHover(mesh: THREE.Object3D | null): void {
   646→    if (this.hovered === mesh) return;
   647→    if (this.hovered) {
   648→      setEmissiveBoost(this.hovered, this.hovered.userData.grabbed ? 0.9 : null);
   649→    }
   650→    this.hovered = mesh;
   651→    if (this.hovered) setEmissiveBoost(this.hovered, 0.85);
   652→  }
   653→
   654→  markGrabbed(mesh: THREE.Object3D | null): void {
   655→    for (const obj of this.grabbables) {
   656→      obj.userData.grabbed = obj === mesh;
   657→      if (obj !== this.hovered) {
   658→        setEmissiveBoost(obj, obj === mesh ? 0.9 : null);
   659→      }
   660→    }
   661→  }

   663→  /** Move the 3D cursor to a world position and recolor by mode. */
   664→  setCursor(worldPos: THREE.Vector3 | null, mode: CursorMode, sizeScale = 1, closeness = 0): void {
   665→    if (!worldPos) {
   666→      this.cursor.visible = false;
   667→      this.cursorRing.visible = false;
   668→      this.rayLine.visible = false;
   669→      return;
   670→    }
   671→    this.cursor.visible = true;
   672→    this.cursor.position.copy(worldPos);
   673→    this.cursor.scale.setScalar(sizeScale);
   674→    // Progress ring cinches from wide/faint to tight/bright with the pinch.
   675→    const c = Math.min(1, Math.max(0, closeness));
   676→    this.cursorRing.visible = true;
   677→    this.cursorRing.position.copy(worldPos);
   678→    this.cursorRing.lookAt(this.camera.position);
   679→    this.cursorRing.scale.setScalar(pinchRingScale(c));
   680→    this.cursorRingMat.opacity = 0.22 + 0.68 * c;
   681→    this.cursorRingMat.color.copy(this.ringCyan).lerp(this.ringMagenta, c);
   682→    const colors: Record<CursorMode, number> = {
   683→      hidden: 0x9adcff,
   684→      point: 0x9adcff,
   685→      hover: 0xa8ffc9,
   686→      pinch: 0xffa8d8,
   687→      grab: 0xffd2a8,
   688→    };
   689→    this.cursorMat.color.setHex(colors[mode]);
   690→    (this.rayLine.material as THREE.LineBasicMaterial).color.setHex(colors[mode]);
   691→
   692→    this.tmpVec.copy(worldPos).sub(this.camera.position).normalize();
   693→    this.rayPositions[0] = this.camera.position.x;
   694→    this.rayPositions[1] = this.camera.position.y;
   695→    this.rayPositions[2] = this.camera.position.z;
   696→    this.rayPositions[3] = worldPos.x + this.tmpVec.x * 10;
   697→    this.rayPositions[4] = worldPos.y + this.tmpVec.y * 10;
   698→    this.rayPositions[5] = worldPos.z + this.tmpVec.z * 10;
   699→    (this.rayLine.geometry.getAttribute('position') as THREE.BufferAttribute).needsUpdate = true;
   700→    this.rayLine.visible = true;
   701→  }

   703→  update(dt: number, elapsed: number): void {
   704→    // Honor cinematic camera override from worlds (e.g. blackhole detonation).
   705→    const cam = this.api?.cinematicCamera;
   706→    if (cam) {
   707→      this.rig.yaw = cam.yaw;
   708→      this.rig.pitch = cam.pitch;
   709→      this.rig.distance = cam.distance;
   710→    }
   711→    this.rig.update(dt, this.camera);
   712→    this.api?.update(dt, elapsed);
   713→    this.cursor.rotation.y += dt * 2.2;
   714→    // Drive per-frame shader uniforms. Starfield twinkle needs elapsed time
   715→    // (seconds, monotonically increasing). Runs on every tier.
   716→    this.starMat.uniforms.uTime.value = elapsed;
   717→  }

   719→  render(): void {
   720→    if (this.quality === 'high' || this.quality === 'ultra') {
   721→      this.composer.render();
   722→    } else {
   723→      this.renderer.render(this.scene, this.camera);
   724→    }
   725→  }

   736→    disposeGroup(this.world);
   737→    disposeGroup(this.labelLayer);
   738→    // Free starfield GPU resources (geometry attributes + shader program).
   739→    this.stars?.geometry.dispose();
   740→    this.starMat?.dispose();
   741→    // Free the fullscreen-quad shader materials backing the post passes.
   742→    (this.caPass?.material as THREE.ShaderMaterial | undefined)?.dispose();
   743→    this.composer?.dispose();
   744→    this.renderer.dispose();
   745→  }
